### Model Test Example

```dart
// test/unit/models/baby_model_test.dart

import 'package:flutter_test/flutter_test.dart';
import 'package:baby_track_flutter/models/baby_model.dart';

void main() {
  group('BabyModel', () {
    group('fromJson', () {
      test('should create BabyModel from valid JSON', () {
        // Arrange
        final json = {
          'id': 'baby_123',
          'name': 'Emma',
          'dateOfBirth': '2025-06-15T00:00:00.000Z',
          'gender': 'female',
          'photoUrl': 'https://example.com/photo.jpg',
        };
        
        // Act
        final baby = BabyModel.fromJson(json);
        
        // Assert
        expect(baby.id, 'baby_123');
        expect(baby.name, 'Emma');
        expect(baby.gender, 'female');
      });
      
      test('should handle null optional fields', () {
        final json = {
          'id': 'baby_123',
          'name': 'Emma',
          'dateOfBirth': '2025-06-15T00:00:00.000Z',
        };
        
        final baby = BabyModel.fromJson(json);
        
        expect(baby.photoUrl, isNull);
        expect(baby.gender, isNull);
      });
    });
    
    group('toJson', () {
      test('should convert BabyModel to valid JSON', () {
        final baby = BabyModel(
          id: 'baby_123',
          name: 'Emma',
          dateOfBirth: DateTime(2025, 6, 15),
        );
        
        final json = baby.toJson();
        
        expect(json['id'], 'baby_123');
        expect(json['name'], 'Emma');
      });
    });
    
    group('age calculation', () {
      test('should calculate age in months correctly', () {
        final baby = BabyModel(
          id: 'baby_123',
          name: 'Emma',
          dateOfBirth: DateTime.now().subtract(Duration(days: 90)),
        );
        
        expect(baby.ageInMonths, 3);
      });
    });
    
    group('vitals validation', () {
      test('should detect abnormal heart rate', () {
        final baby = BabyModel(
          id: 'baby_123',
          name: 'Emma',
          dateOfBirth: DateTime.now().subtract(Duration(days: 90)),
          currentHeartRate: 200, // Abnormal
        );
        
        expect(baby.hasAbnormalVitals, isTrue);
      });
    });
  });
}
```

### Service Test Example

```dart
// test/unit/services/auth_service_test.dart

import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';
import 'package:firebase_auth/firebase_auth.dart' as firebase;
import 'package:baby_track_flutter/services/auth_service.dart';

// Mocks
class MockFirebaseAuth extends Mock implements firebase.FirebaseAuth {}
class MockUserCredential extends Mock implements firebase.UserCredential {}
class MockUser extends Mock implements firebase.User {}

void main() {
  late AuthService authService;
  late MockFirebaseAuth mockFirebaseAuth;
  
  setUp(() {
    mockFirebaseAuth = MockFirebaseAuth();
    authService = AuthService(firebaseAuth: mockFirebaseAuth);
  });
  
  group('AuthService', () {
    group('signInWithEmail', () {
      test('should return user on successful sign in', () async {
        // Arrange
        final mockCredential = MockUserCredential();
        final mockUser = MockUser();
        
        when(() => mockUser.uid).thenReturn('user_123');
        when(() => mockCredential.user).thenReturn(mockUser);
        when(() => mockFirebaseAuth.signInWithEmailAndPassword(
          email: any(named: 'email'),
          password: any(named: 'password'),
        )).thenAnswer((_) async => mockCredential);
        
        // Act
        final result = await authService.signInWithEmail(
          email: 'test@example.com',
          password: 'password123',
        );
        
        // Assert
        expect(result.isSuccess, isTrue);
        expect(result.user?.uid, 'user_123');
      });
      
      test('should return error on invalid credentials', () async {
        when(() => mockFirebaseAuth.signInWithEmailAndPassword(
          email: any(named: 'email'),
          password: any(named: 'password'),
        )).thenThrow(
          firebase.FirebaseAuthException(
            code: 'wrong-password',
            message: 'Invalid password',
          ),
        );
        
        final result = await authService.signInWithEmail(
          email: 'test@example.com',
          password: 'wrongpassword',
        );
        
        expect(result.isSuccess, isFalse);
        expect(result.errorCode, 'wrong-password');
      });
    });
  });
}
```

```dart
// test/widget/widgets/vital_card_test.dart

import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:baby_track_flutter/widgets/vital_card.dart';

void main() {
  group('VitalCard', () {
    testWidgets('displays vitals correctly', (tester) async {
      await tester.pumpWidget(
        MaterialApp(
          home: Scaffold(
            body: VitalCard(
              heartRate: 120,
              temperature: 36.8,
              oxygenLevel: 98,
            ),
          ),
        ),
      );
      
      expect(find.text('120'), findsOneWidget);
      expect(find.text('36.8°'), findsOneWidget);
      expect(find.text('98%'), findsOneWidget);
    });
    
    testWidgets('shows warning for abnormal heart rate', (tester) async {
      await tester.pumpWidget(
        MaterialApp(
          home: Scaffold(
            body: VitalCard(
              heartRate: 200, // Abnormal
              temperature: 36.8,
              oxygenLevel: 98,
            ),
          ),
        ),
      );
      
      // Should show warning indicator
      expect(find.byIcon(Icons.warning), findsOneWidget);
    });
    
    testWidgets('triggers callback on tap', (tester) async {
      var tapped = false;
      
      await tester.pumpWidget(
        MaterialApp(
          home: Scaffold(
            body: VitalCard(
              heartRate: 120,
              onTap: () => tapped = true,
            ),
          ),
        ),
      );
      
      await tester.tap(find.byType(VitalCard));
      
      expect(tapped, isTrue);
    });
    
    testWidgets('has correct semantics for accessibility', (tester) async {
      await tester.pumpWidget(
        MaterialApp(
          home: Scaffold(
            body: VitalCard(
              heartRate: 120,
              temperature: 36.8,
              oxygenLevel: 98,
            ),
          ),
        ),
      );
      
      expect(
        tester.getSemantics(find.byType(VitalCard)),
        matchesSemantics(label: 'Vitals: Heart rate 120, Temperature 36.8, Oxygen 98%'),
      );
    });
  });
}
```

```dart
// integration_test/auth_flow_test.dart

import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:baby_track_flutter/main.dart' as app;

void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  
  group('Authentication Flow', () {
    testWidgets('complete sign up journey', (tester) async {
      app.main();
      await tester.pumpAndSettle();
      
      // Welcome screen
      expect(find.text('Welcome to BabyTrack'), findsOneWidget);
      await tester.tap(find.text('Get Started'));
      await tester.pumpAndSettle();
      
      // Phone entry
      expect(find.text('Enter your phone number'), findsOneWidget);
      await tester.enterText(
        find.byType(TextField),
        '+919876543210',
      );
      await tester.tap(find.text('Continue'));
      await tester.pumpAndSettle();
      
      // OTP screen (use test OTP: 123456)
      expect(find.text('Enter OTP'), findsOneWidget);
      for (var i = 0; i < 6; i++) {
        await tester.enterText(
          find.byKey(Key('otp_$i')),
          '${i + 1}',
        );
      }
      await tester.pumpAndSettle();
      
      // PIN setup
      expect(find.text('Create a PIN'), findsOneWidget);
      for (var i = 0; i < 4; i++) {
        await tester.tap(find.text('${i + 1}'));
      }
      await tester.pumpAndSettle();
      
      // Confirm PIN
      for (var i = 0; i < 4; i++) {
        await tester.tap(find.text('${i + 1}'));
      }
      await tester.pumpAndSettle();
      
      // Should be on dashboard
      expect(find.text('Dashboard'), findsOneWidget);
    });
    
    testWidgets('biometric login flow', (tester) async {
      // Requires previous sign up
      app.main();
      await tester.pumpAndSettle();
      
      // Should show PIN entry for returning user
      expect(find.text('Enter your PIN'), findsOneWidget);
      
      // Tap biometric button
      await tester.tap(find.byIcon(Icons.fingerprint));
      await tester.pumpAndSettle();
      
      // Biometric prompt would appear (mocked in test)
      // Should be on dashboard after success
      expect(find.text('Dashboard'), findsOneWidget);
    });
  });
}
```

## 10. Test Maintenance

### Best Practices

1. **Test naming**: Use descriptive names following pattern:
   ```dart
   test('should [expected behavior] when [condition]', () {});
   ```

2. **Arrange-Act-Assert**: Structure all tests with clear sections
   ```dart
   test('should return user on successful login', () {
     // Arrange
     final mockAuth = MockFirebaseAuth();
     
     // Act
     final result = authService.login(email, password);
     
     // Assert
     expect(result.isSuccess, isTrue);
   });
   ```

3. **One assertion per test** (when practical)

4. **Don't test implementation details** - test behavior

5. **Keep tests fast** - mock external dependencies

6. **Update tests with code changes** - tests are first-class code
